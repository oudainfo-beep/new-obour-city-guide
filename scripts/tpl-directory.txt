const directories = fs.readdirSync(path.join(root, "data", "directories"))
  .filter((f) => f.endsWith(".json") && f !== "index.json")
  .map((f) => JSON.parse(fs.readFileSync(path.join(root, "data", "directories", f), "utf8")));
const dirBySlug = Object.fromEntries(directories.map((d) => [d.slug, d]));

const mapsUrl = (it) => `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(`${it.n} ${it.a || "مدينة العبور"}`)}`;

function dirItem(it) {
  const actions = [
    it.t ? `<a class="dir-call" href="tel:${it.t}">☎ ${it.p}</a>` : "",
    `<a href="${mapsUrl(it)}" target="_blank" rel="noopener noreferrer">الخريطة ↗</a>`,
  ].filter(Boolean).join("");
  return `<article class="dir-item"><h4>${it.n}</h4>${it.e && it.e !== it.n ? `<p class="dir-en">${it.e}</p>` : ""}${it.a ? `<p class="dir-addr">⌖ ${it.a}</p>` : ""}<div class="dir-actions">${actions}</div>${it.s ? `<small>المصدر: ${it.s}</small>` : ""}</article>`;
}

function dirGroups(d) {
  if (!d.subs.length) return `<div class="dir-list">${d.items.map(dirItem).join("")}</div>`;
  const chips = d.subs.map((s, i) => `<a href="#g${i}">${s} <b>${d.items.filter((it) => it.c === s).length}</b></a>`).join("");
  const groups = d.subs.map((s, i) => {
    const items = d.items.filter((it) => it.c === s);
    return `<section class="dir-group" id="g${i}"><h3>${s}<small>${items.length}</small></h3><div class="dir-list">${items.map(dirItem).join("")}</div></section>`;
  }).join("");
  const rest = d.items.filter((it) => !d.subs.includes(it.c));
  const restHtml = rest.length ? `<section class="dir-group" id="gx"><h3>أخرى<small>${rest.length}</small></h3><div class="dir-list">${rest.map(dirItem).join("")}</div></section>` : "";
  return `<nav class="dir-chips" aria-label="أقسام الدليل">${chips}</nav>${groups}${restHtml}`;
}
